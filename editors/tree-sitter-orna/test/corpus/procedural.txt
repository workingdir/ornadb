CREATE SERVER FUNCTION tasks.complete (p_task REF tasks.task)
RETURNS REF tasks.task
SECURITY INVOKER
TRANSACTION ATOMIC
IS
    LET v_now TIMESTAMP := sys.time.now();
    CONST v_grace INT := 5;
    STATE counter INT SCOPE SESSION DEFAULT 0;
BEGIN
    v_now := sys.time.now();
    LET v_later := v_now;
    CALL tasks.touch(p_task);
    AWAIT sys.io.flush();
    IF v_later > v_now THEN
        RETURN p_task;
    ELSIF v_now IS NULL THEN
        RETURN NULL;
    ELSE
        v_now := sys.time.now();
    END IF;
    WHILE counter < v_grace LOOP
        counter := counter + 1;
    END LOOP;
    FOR t IN tasks.pending() LOOP
        CALL tasks.touch(t);
    END LOOP;
    RETURN p_task;
END;
