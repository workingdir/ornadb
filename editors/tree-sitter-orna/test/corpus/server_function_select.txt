CREATE SERVER FUNCTION tasks.overdue (
    p_principal REF sys.security.principal,
    p_before TIMESTAMP DEFAULT sys.time.now()
)
RETURNS TABLE (
    task    REF tasks.task,
    title   TEXT
)
SECURITY INVOKER
TRANSACTION READ ONLY
VOLATILITY STABLE
AS
    SELECT
        REF(t),
        t.title
    FROM tasks.task t
    WHERE t.completed = FALSE
      AND t.due_at < p_before
    ORDER BY t.due_at;
