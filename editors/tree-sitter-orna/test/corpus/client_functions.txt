CREATE CLIENT FUNCTION tasks.overdue_window()
RETURNS std.ui.UI
IS
    STATE filter TEXT
        SCOPE USER
        DEFAULT '';

    LET rows std.data.Resource<TABLE(
        task REF tasks.task,
        title TEXT
    )> := std.data.resource(tasks.overdue);
BEGIN
    RETURN std.ui.window(
        title => 'Overdue Tasks',
        content => std.ui.column(
            children => [
                std.ui.text_input(
                    value       => filter,
                    placeholder => 'Filter tasks',
                    on_change   => filter.SET
                )
            ]
        )
    );
END;
