CREATE CLIENT FUNCTION demo.calc(p_x INT, p_y TEXT)
RETURNS TEXT
AS
    std.strings.concat(
        p_y,
        std.strings.to_text(p_x + 2 * 3 - 4 % 2 || ' suffix'),
        suffix => '!'
    ) || CASE WHEN p_x > 0 THEN 'pos' ELSE 'neg' END;

RETURNS TEXT
AS
    SELECT
        'plain' || 'it''s escaped',
        DATE '2024-01-15',
        TIMESTAMP '2024-01-15 10:30:00',
        BYTES '\x00\x01',
        a.b.c,
        TRUE,
        FALSE,
        NULL,
        42,
        3.14,
        -1,
        NOT TRUE,
        x IS NOT NULL,
        y IN (1, 2),
        z LIKE 'a%',
        w ILIKE 'b%';
